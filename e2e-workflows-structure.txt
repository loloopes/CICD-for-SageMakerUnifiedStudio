# ---------------------------------------------------------------------------
# 5) What each entry workflow uses (full capacity)
# ---------------------------------------------------------------------------
# e2e-dataops-pipeline.yml
#   → smus-direct-deploy.yml
#   → dataops-pipeline/** + src/smus_cicd
#   → run data_pipeline (refresh transformed/ + Athena)
#
# e2e-mlops-pipeline.yml
#   → setup-mlops-infra.sh            (EventBridge/Lambda approve trigger)
#   → smus-direct-deploy.yml          (dev→test→prod, setup_mlflow=true)
#   → drift_check_pipeline            (check_drift=true)
#   → e2e-mlops-train.yml             (on drift OR run_training=true)
#       → training_pipeline           (Clarify + Model Card + register)
#
# e2e-mlops-promote.yml
#   → mlops_helper.py (select + stage artifacts)
#   → await_issue_approval.sh         (TEST + PROD gates; Issues enabled)
#   → deploy_pipeline per stage + smoke-test + package metadata
#   ← also triggered by repository_dispatch: model-approved
#
# e2e-mlops-batch-inference.yml
#   → mlops_helper.py resolve-batch-context
#   → batch_inference_smoke.py        (dev/test/prod endpoints)
#   → schedule via BATCH_INFERENCE_CRON (mirror cron in YAML)

# ---------------------------------------------------------------------------
# 6) GitHub Environments / secrets / vars (full capacity)
# ---------------------------------------------------------------------------
# Environments (required):
#   dev-aws-account
#   test-aws-account
#   prod-aws-account
#
# Secrets (required):
#   AWS_ROLE_ARN_DEV
#   AWS_ROLE_ARN_TEST                 # promote + batch at full multi-stage
#   AWS_ROLE_ARN_PROD
#   PROJECT_OWNER_ARN
#
# Vars — set these for full capacity (defaults exist but production use sets them):
#   DOMAIN_REGION                     # required in practice
#   DEV_PROJECT_NAME                  # e2e-data-stack-ml-ops-dev
#   TEST_PROJECT_NAME                 # e2e-data-stack-ml-ops-test
#   PROD_PROJECT_NAME                 # e2e-data-stack-ml-ops-prod
#   DOMAIN_TAG_PURPOSE                # domain discovery for stage-artifact / batch
#   ATHENA_DB_NAME                    # dataops Athena DB
#   MODEL_PACKAGE_GROUP               # stack-prediction-models
#   MLFLOW_TRACKING_SERVER_NAME       # smus-direct-deploy setup_mlflow
#   MLFLOW_TRACKING_SERVER_ARN        # optional explicit ARN
#   DEPLOY_TRIGGER_FUNCTION_NAME      # stack-airflow-deploy-trigger
#   DEPLOY_TRIGGER_RULE_NAME          # stack-model-approved
#   DEPLOY_TRIGGER_ROLE_NAME          # stack-airflow-deploy-trigger-role
#   BATCH_INFERENCE_CRON              # e.g. '0 14 * * 1' (keep YAML cron in sync)
#   BATCH_SHARED_S3_URI               # optional override
#   BATCH_FEATURES_SHARED_S3_URI      # optional override (cross-stage features)
#   PROJECT_EXECUTION_ROLE_ARN        # optional batch override
#
# Repo settings:
#   Issues ENABLED                    # promote approval gates
#   Actions OIDC trust via setup-github-oidc.sh (one-time)
